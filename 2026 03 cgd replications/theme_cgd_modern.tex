\newcommand{\lightsource}[1]{%

  {\raggedleft\usebeamerfont{source line}\color{cgdmuted}Source: \url{#1}\par}
}

\newcommand{\sectioncardbody}[2]{%
  \begin{tikzpicture}[remember picture, overlay]
    \fill[cgdblue] (current page.south west) rectangle (current page.north east);
  \end{tikzpicture}
  \vspace*{0.08\textheight}
  \begin{center}
    {\usebeamerfont{section title}\color{cgdsand}#1\par}
    \ifstrempty{#2}{}{%
      \vspace{0.28cm}
      {\Large\color{cgdsand!90}#2\par}
    }%
  \end{center}
}

\newcommand{\darkimageframebody}[3]{%
  \begin{tikzpicture}[remember picture, overlay]
    \fill[cgdblue] (current page.south west) rectangle (current page.north east);
  \end{tikzpicture}
  {\usebeamerfont{frametitle}\color{cgdsand}#1\par}
  \vspace{0.12cm}
  \begin{center}
    \includegraphics[
      width=0.93\textwidth,
      height=0.74\textheight,
      keepaspectratio
    ]{#2}
  \end{center}
  \vfill
  {\raggedleft\usebeamerfont{source line}\color{cgdsand!82}%
    Source: \url{#3}\par}
}
